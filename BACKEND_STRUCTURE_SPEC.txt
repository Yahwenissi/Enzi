BACKEND STRUCTURE SPECIFICATION - ENZI
======================================

This document outlines the data layer and backend structure for the Enzi project.
It is structured to allow a backend developer to implement Prisma models, APIs,
auth, and database seed data independently of the frontend.

=== 1. DATABASE SCHEMA (prisma/schema.prisma) ===

// Core models for better-auth + Enzi domain

model User {
  id            String    @id @default(cuid())
  email         String    @unique
  emailVerified Boolean   @default(false)
  name          String?
  image         String?
  role          Role      @default(USER)
  level         Int       @default(1)
  xp            Int       @default(0)
  createdAt     DateTime  @default(now())
  updatedAt     DateTime  @updatedAt

  // better-auth relations
  accounts      Account[]
  sessions      Session[]
  verifications Verification[]

  // Enzi domain
  spots         Spot[]
  bookings      Booking[]
  reviews       Review[]
  badges        UserBadge[]
  visits        Visit[]
}

model Spot {
  id          String   @id @default(cuid())
  name        String
  description String?
  category    Category @relation(fields: [categoryId], references: [id])
  categoryId  String
  address     String
  latitude    Float    // WGS84 for gebeta
  longitude   Float
  phone       String?
  website     String?
  hours       Json?    // { mon: "9-22", tue: "9-22", ... }
  // See section 17. Attribution is mandatory for CC BY / CC BY-SA imagery, so
  // images are objects rather than bare URLs.
  images      Json?    // [{ url, author?, license?, sourceUrl? }]
  rating      Float    @default(0)
  reviewCount Int      @default(0)
  isVerified  Boolean  @default(false)
  createdAt   DateTime @default(now())
  updatedAt   DateTime @updatedAt

  creator     User     @relation(fields: [creatorId], references: [id])
  creatorId   String
  bookings    Booking[]
  reviews     Review[]
  visits      Visit[]
}

model Category {
  id          String @id @default(cuid())
  name        String @unique
  slug        String @unique
  icon        String // lucide icon name
  color       String // tailwind color class
  description String?
  spots       Spot[]
}

// userId is nullable to support guest bookings: the product promise is "Book in
// two taps", so an anonymous visitor must be able to reserve without an account.
// A guest booking is claimed by setting userId when the same browser later signs
// in. Once auth lands, decide whether guest rows should expire instead.
model Booking {
  id        String        @id @default(cuid())
  user      User?         @relation(fields: [userId], references: [id], onDelete: Cascade)
  userId    String?
  spot      Spot          @relation(fields: [spotId], references: [id], onDelete: Cascade)
  spotId    String
  status    BookingStatus @default(PENDING)
  date      DateTime
  partySize Int
  notes     String?
  // Guest contact details, required only while userId is null.
  guestName String?
  guestEmail String?
  createdAt DateTime      @default(now())
  updatedAt DateTime      @updatedAt

  @@index([spotId, date])
  @@index([userId])
}

model Review {
  id        String   @id @default(cuid())
  user      User     @relation(fields: [userId], references: [id])
  userId    String
  spot      Spot     @relation(fields: [spotId], references: [id])
  spotId    String
  rating    Int      // 1-5
  comment   String?
  createdAt DateTime @default(now())
  updatedAt DateTime @updatedAt
}

model Badge {
  id          String @id @default(cuid())
  name        String @unique
  description String
  icon        String
  criteria    Json   // { type: "VISITS", count: 10, categoryId?: "parks" }
  users       UserBadge[]
}

model UserBadge {
  id     String @id @default(cuid())
  user   User   @relation(fields: [userId], references: [id])
  userId String
  badge  Badge  @relation(fields: [badgeId], references: [id])
  badgeId String
  earnedAt DateTime @default(now())
  @@unique([userId, badgeId])
}

// Append-only visit log.
//
// The earlier `@@unique([userId, spotId])` constraint was removed: it permitted
// exactly one visit per user per spot forever, which caps lifetime XP and makes
// the "level up as you go" progression impossible. Visits are now append-only
// so a user can log real, repeated history.
//
// Anti-spam is enforced in application code instead of the database -- see
// section 16 for the XP award rules.
model Visit {
  id        String   @id @default(cuid())
  user      User     @relation(fields: [userId], references: [id], onDelete: Cascade)
  userId    String
  spot      Spot     @relation(fields: [spotId], references: [id], onDelete: Cascade)
  spotId    String
  visitedAt DateTime @default(now())
  xpAwarded Int      @default(0) // 0 when logged inside the daily cooldown
  createdAt DateTime @default(now())

  @@index([userId, spotId, visitedAt])
  @@index([userId, visitedAt])
}

// better-auth required models

model Account {
  id                String  @id @default(cuid())
  accountId         String
  providerId        String
  userId            String
  accessToken       String?
  refreshToken      String?
  idToken           String?
  accessTokenExpiresAt DateTime?
  refreshTokenExpiresAt DateTime?
  scope             String?
  password          String?
  createdAt         DateTime @default(now())
  updatedAt         DateTime @updatedAt
  user              User     @relation(fields: [userId], references: [id], onDelete: Cascade)
  @@unique([providerId, accountId])
}

model Session {
  id           String   @id @default(cuid())
  expiresAt    DateTime
  token        String   @unique
  createdAt    DateTime @default(now())
  updatedAt    DateTime @updatedAt
  ipAddress    String?
  userAgent    String?
  userId       String
  user         User     @relation(fields: [userId], references: [id], onDelete: Cascade)
}

model Verification {
  id        String   @id @default(cuid())
  identifier String
  value     String
  expiresAt DateTime
  createdAt DateTime @default(now())
  updatedAt DateTime @updatedAt
  @@unique([identifier, value])
}

// === 16. VISIT XP RULES (ANTI-SPAM) ===

// Because Visit is append-only (section 1), the database no longer prevents a
// user from logging the same spot repeatedly. Abuse prevention therefore lives
// in application code, and MUST live there -- the API is the only place that
// knows the request is genuine.
//
// Chosen mechanic: DAILY COOLDOWN. A visit is always persisted so the user
// keeps an honest history record, but XP is awarded only for the first visit to
// a given spot on a given calendar day. Repeat visits on the same day are
// stored with xpAwarded = 0.
//
// Rejected alternatives, for the record:
//   - Diminishing returns (50/10/0 XP): still gameable by spreading across many
//     spots, and the taper curve has to be tuned forever.
//   - Global daily XP cap: simple, but it lets one spot be farmed to the cap
//     before the user moves on.
//
// POST /api/visits - Record a visit (auth required)

const XP_PER_VISIT = 50
const VISITS_PER_SPOT_PER_DAY = 1 // "first visit per spot per day earns XP"

// Start of the current calendar day in the user's timezone.
function startOfLocalDay(now = new Date()): Date {
  return new Date(now.getFullYear(), now.getMonth(), now.getDate())
}

export async function recordVisit(userId: string, spotId: string) {
  const since = startOfLocalDay()

  // Visits already logged for this spot today.
  const todaysVisits = await prisma.visit.findMany({
    where: { userId, spotId, visitedAt: { gte: since } },
    select: { id: true },
  })

  const xpAwarded =
    todaysVisits.length < VISITS_PER_SPOT_PER_DAY ? XP_PER_VISIT : 0

  // Persist regardless of cooldown: the history stays complete.
  const visit = await prisma.visit.create({
    data: { userId, spotId, visitedAt: new Date(), xpAwarded },
  })

  if (xpAwarded > 0) {
    await prisma.user.update({
      where: { id: userId },
      data: {
        xp: { increment: xpAwarded },
        // Recompute level rather than incrementing it, so the curve can change
        // without stranding users on a stale level.
        level: levelForXp(currentXp + xpAwarded),
      },
    })
  }

  // xpAwarded is returned so the client can show "visit logged" vs "+50 XP".
  return { visit, xpAwarded }
}

// Level curve, e.g. level = floor(sqrt(xp / 100)) + 1
export function levelForXp(xp: number): number {
  return Math.floor(Math.sqrt(xp / 100)) + 1
}

// Badges must also be evaluated against *distinct* spots, otherwise a user
// farming one spot could satisfy a "visit N spots" badge via the cooldown
// loophole. Count distinct spotId among visits that actually earned XP.
const badgeEarnedVisits = await prisma.visit.findMany({
  where: { userId, xpAwarded: { gt: 0 } },
  select: { spotId: true },
  distinct: ["spotId"],
})

// === 17. SPOT IMAGERY ===
//
// Spot.images is Json and holds objects, not bare URLs:
//   { url, author?, license?, sourceUrl? }
//
// Attribution is a licensing obligation, not a nicety: CC BY and CC BY-SA both
// require crediting the author. Storing bare URLs makes that impossible to
// satisfy at render time, so the credit travels with the URL.
//
// Seed imagery is sourced from Wikimedia Commons, where each file carries an
// explicit free license (CC0 / CC BY / CC BY-SA / public domain).
//
// REQUIREMENTS for whoever writes the seeder:
//   - Record license + author for every image, and set sourceUrl to the Commons
//     file description page. The detail page renders the credit from this.
//   - Do not attach an image to a spot unless the photo actually shows that
//     spot. Text search returns confident wrong answers -- searching "Friendship
//     Square" returns photographs of Friendship Square in Dalian, China, and
//     "Ambassador Theatre" returns New York City. Verify each match by eye
//     before seeding it.
//   - A spot with no verified photo is fine. Leave images empty and let the UI
//     render its placeholder; do not substitute a generic cityscape.
//   - Pin the exact Commons file used so re-seeding is reproducible and a later
//     takedown has a known blast radius.
//   - The frontend pins next.config.ts images.remotePatterns. If seeding starts
//     using a different host, extend that list or next/image will refuse to
//     optimise those URLs at runtime.
